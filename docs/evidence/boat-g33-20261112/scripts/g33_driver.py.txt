def scrub(text: str) -> str:
    for s in SECRETS:
        if s:
            text = text.replace(s, "[redacted]")
    return KEY_SHAPED.sub("[redacted]", text)


def emit(**fields: object) -> None:
    line = scrub(json.dumps(fields, sort_keys=True, default=str))
    RECEIPT.write(line + "\n")
    RECEIPT.flush()
    print(line, flush=True)


def judge(ok: object, what: str) -> bool:
    if not ok:
        FAILS.append(what)
        emit(step="FAIL", what=what)
    return bool(ok)


def body(r: httpx.Response) -> dict:
    try:
        d = r.json()
    except ValueError:
        return {}
    return d if isinstance(d, dict) else {}


def boat(key: str, org: str = ORG) -> httpx.Client:
    return httpx.Client(base_url=BOAT_API, timeout=60.0,
                        headers={"Authorization": f"Bearer {key}", "X-Boat-Org": org})


def live_ws(prod: httpx.Client) -> dict[str, str]:
    """antiek-ws- sandboxes this key can list, id -> state (ids and states only: never urls)."""
    rows = body(prod.get("/sandboxes", params={"namePrefix": "antiek-ws-"})).get("sandboxes") or []
    return {str(x.get("id")): str(x.get("state")) for x in rows
            if isinstance(x, dict) and str(x.get("name", "")).startswith("antiek-ws-")
            and str(x.get("state")) not in GONE_STATES}


def ts(text: str) -> float:
    return datetime.strptime(text, "%Y-%m-%d %H:%M:%S,%f").replace(tzinfo=timezone.utc).timestamp()


def limits(prod: httpx.Client, step: str) -> dict:
    lim = body(prod.get("/limits"))
    emit(step=step, teamId=lim.get("teamId"), canStart=lim.get("canStart"),
         activeSandboxes=lim.get("activeSandboxes"), nextStartDollars=lim.get("nextStartDollars"),
         starts=lim.get("starts"), vmHoursRemaining=(lim.get("planIncluded") or {}).get("vmHoursRemaining"),
         wallet=(lim.get("wallet") or {}).get("balanceDollars"))
    return lim

    judge(states == ["done"] * 3, "a leaf did not end done")

def stop_service(ctx: dict) -> None:
    svc = ctx.get("svc")
    if svc is None or svc.poll() is not None:
        return
    svc.send_signal(signal.SIGINT)
    try:
        svc.wait(timeout=60)
    except subprocess.TimeoutExpired:
        svc.kill()
        svc.wait()
    emit(step="service_stopped", rc=svc.returncode)


def verify_log(ctx: dict) -> None:
    path = f"{T}/logs/service.log"
    lines = open(path, encoding="utf-8", errors="replace").read().splitlines() if os.path.exists(path) else []
    sel = ready = 0
    acc: dict[str, dict] = {}
    made: dict[str, dict] = {}
    gath: dict[str, dict] = {}
    gone: dict[str, float] = {}
    gone_n: dict[str, int] = {}
    warn = manual = shaped = 0
    for n, raw in enumerate(lines):
        shaped += bool(KEY_SHAPED.search(raw))
        m = LINE.match(raw)
        if not m:
            continue
        t, level, name, msg = ts(m[1]), m[2], m[3], m[4]
        warn += level in ("WARNING", "ERROR", "CRITICAL") and name.startswith("antiek")
        manual += "Delete it manually" in msg
        sel += msg == SELECTED
        ready += bool(READY.search(msg))
        if (g := ACC.search(msg)) and g[1] not in acc:
            acc[g[1]] = {"ws": g[2], "type": g[3], "ttl": g[4], "createdAt": g[5], "via": g[6], "t": t, "n": n}
        if g := CREATED.search(msg):
            made[g[1]] = {"bx": g[2], "type": g[4], "ttl": g[5], "env": g[6], "net": g[7], "uid": int(g[8])}
        if g := GATHER.search(msg):
            gath[g[3]] = {"iid": g[1], "backend": g[2], "passes": int(g[4]), "uid": int(g[5])}
        if (g := DELETED.search(msg)) and g[3] == "destroy" and g[4] == "deleted":
            gone[g[1]] = t
            gone_n.setdefault(g[1], n)
    ctx.update(acc=acc, made=made, gone=gone)
    leaf_ids = {f"{ctx.get('sid')}-leaf-{i}" for i in range(3)}
    emit(step="journal", selected=sel, ready=ready, accepted=len(acc), created=len(made), gathered=len(gath),
         deleted=len(gone), antiek_warnings=warn, key_shaped_lines=shaped,
         ledger=sorted(acc), workspaces=sorted(a["ws"] for a in acc.values()))
    if not ctx.get("sid"):
        judge(not acc, "sandboxes were accepted although no launch succeeded")
        return
    judge(sel == 1 and ready == 1, "selected/ready lines")
    judge(len(acc) == 3 and all(a["via"] == "create" and a["type"] == "small" and a["ttl"] == str(TTL_S)
                                for a in acc.values()), "three accepted via=create small 900")
    judge({a["ws"] for a in acc.values()} == set(made) == set(gath), "workspace ids do not join")
    judge(all(made[a["ws"]]["bx"] == bx for bx, a in acc.items() if a["ws"] in made), "sandbox ids do not join")
    judge(all(w["env"] == ENV_NAME and w["net"] == "deny_all" and w["uid"] != 0 and w["type"] == "small"
              for w in made.values()), "created lines: environment, network or uid")
    judge({g["iid"] for g in gath.values()} == leaf_ids and all(
        g["backend"] == "boat" and g["passes"] == 2 and g["uid"] == made.get(w, {}).get("uid")
        for w, g in gath.items()), "gather lines")
    judge(set(gone) == set(acc), "a deleted line is missing for an accepted id")
    # All three alive at once: the last accepted line precedes the first deleted line.
    judge(acc and gone_n and max(a["n"] for a in acc.values()) < min(gone_n.values()), "lifetimes did not overlap")
    judge(manual == 0 and shaped == 0, "delete failure or key-shaped text in the service log")


def verify_state(ctx: dict) -> None:
    sid, acc, made = ctx.get("sid"), ctx.get("acc", {}), ctx.get("made", {})
    rows = [json.loads(x) for x in open(f"{ST}/boat_ledger.jsonl", encoding="utf-8") if x.strip()]
    attempts = [x for x in rows if "id" in x]
    binds = {x.get("sandbox") for x in rows if "bind" in x}
    kinds = {k: sum(k in x for x in rows) for k in ("send", "settle", "void")}
    emit(step="boat_ledger", attempts=len(attempts), binds=sorted(binds), other_rows=kinds)
    judge(len(attempts) == 3 and all(x.get("type") == "small" and x.get("ttl") == TTL_S for x in attempts)
          and binds == set(acc) and kinds["void"] == 0, "Boat ledger")
    g = subprocess.run([PY, "-c", GRAPH], cwd=SRC, env=BASE_ENV, capture_output=True, text=True, timeout=120)
    out = json.loads(g.stdout or "{}")
    ins = out.get("insights") or []
    emit(step="graph", rc=g.returncode, **out)
    uid_by_ws = {w: v["uid"] for w, v in made.items()}
    judge(out.get("types") == {"insight": 3, "question": 4}, "node types are not 3 insight + 4 question")
    judge(out.get("step4_verbatim") == 7, "remote-exec-fanout Step 4 query (4 plan questions + 3 insights)")
    judge((out.get("relations") or {}).get("decomposes_into") == 3
          and (out.get("relations") or {}).get("resolved_by") == 3, "edge counts")
    judge(sorted(m.get("investigation_id") for _, m in ins) == [f"{sid}-leaf-{i}" for i in range(3)]
          and all(m.get("source") == "research_runner" and m.get("promoted_kind") == "insight"
                  and m.get("gather_mode") == "exec_backend" and m.get("backend") == "boat"
                  and m.get("contained_passes") == 2 and m.get("workspace_id") in uid_by_ws
                  and m.get("ran_as_uid") == uid_by_ws.get(m.get("workspace_id"))
                  and "origin_event_id" not in m and "source_event_ids" not in m for _, m in ins),
          "insight metadata: not funnel-promoted contained notes")
    rb = out.get("resolved_by") or []
    judge(sorted(s for s, _ in rb) == ctx.get("qids") and sorted(t for _, t in rb) == sorted(n for n, _ in ins),
          "resolved_by edges do not join the launch's question nodes to the insights")
    ev = f"{ST}/research_events"
    for i in range(3):
        evs = [json.loads(x) for x in open(f"{ev}/{sid}-leaf-{i}.jsonl", encoding="utf-8") if x.strip()]
        n_ins = sum(e.get("action_type") == "graph.node.inserted" and (e.get("payload") or {}).get("node_type") == "insight" for e in evs)
        n_done = sum(e.get("action_type") == "investigation.completed" for e in evs)
        emit(step="events", leaf=i, insight_inserted=n_ins, completed=n_done)
        judge(n_ins == 1 and n_done == 1, f"event log of leaf {i}")
    launched = [json.loads(x) for x in open(f"{ev}/{sid}.jsonl", encoding="utf-8") if '"cascade.launched"' in x]
    judge(len(launched) == 1 and (launched[0].get("payload") or {}).get("leaf_count") == 3, "cascade.launched")


def teardown_and_revoke(ctx: dict) -> None:
    prod, acc, gone = ctx.get("prod"), ctx.get("acc", {}), ctx.get("gone", {})
    if prod is not None:
        now = time.time()
        for bx in sorted(acc):
            if bx in PROTECTED:
                judge(False, f"{bx} is a research box: not touched")
                continue
            status = prod.get(f"/sandboxes/{bx}").status_code
            if status != 404:
                prod.delete(f"/sandboxes/{bx}", headers={"X-Ascii-Confirm-Delete": bx})
                status = prod.get(f"/sandboxes/{bx}").status_code
                gone.setdefault(bx, time.time())
            emit(step="gone", sandbox=bx, status=status)
            judge(status == 404, f"{bx} still answers after teardown")
        unlogged = sorted(set(ctx["seen"]) - set(acc))
        judge(not unlogged, f"listed with no accepted line (not deleted; owner decides): {unlogged}")
        after = limits(prod, "limits_after")
        # Recorded, not judged: the research boxes share the org (runbook "After all three parts").
        emit(step="active_delta", before=ctx.get("lim_before", {}).get("activeSandboxes"),
             after=after.get("activeSandboxes"))
        bound = 0.0
        for bx, a in acc.items():
            try:
                start = datetime.fromisoformat(a["createdAt"]).timestamp()
            except ValueError:
                start = a["t"]
            bound += max(0.0, gone.get(bx, now) - start) / 3600 * SMALL_USD_PER_H
        emit(step="cost", list_price_bound_usd=round(bound, 6),
             full_ttl_bound_usd=round(len(acc) * TTL_S / 3600 * SMALL_USD_PER_H, 4), cap_usd=CAP_USD)
        judge(bound < CAP_USD, "cost bound above the G3.3 ceiling")
    if ctx.get("kid"):
        rv = boat(OWNER_KEY).delete(f"/api-keys/{ctx['kid']}").status_code
        dead = boat(ctx["temp"]).get("/limits").status_code
        emit(step="revoke", key_id=ctx["kid"], revoke_status=rv, after_status=dead)
        judge(rv < 300 and dead == 401, "temp key not dead after revoke")


def repo_fixtures() -> set[bytes]:
    """Key-shaped strings committed at the checked-out SHA: the suite's synthetic fixtures.
    They are public by construction, and the hermetic pre-flight echoes them into --basetemp
    (the 2026-11-12 run found exactly these four, nothing else), so they are not leaks."""
    dirty = subprocess.run(["git", "-C", SRC, "status", "--porcelain", "--untracked-files=no"],
                           capture_output=True, text=True, check=True).stdout
    judge(not dirty.strip(), "the source checkout differs from its commit")
    found = subprocess.run(["git", "-C", SRC, "grep", "-I", "-h", "-o", "-E", "-e",
                            KEY_SHAPED.pattern, "HEAD"], capture_output=True, check=False)
    judge(found.returncode in (0, 1), f"git grep for committed fixtures exited {found.returncode}")
    return {line for line in found.stdout.splitlines() if line}


def leak_scan() -> None:
    needles = [s.encode() for s in SECRETS if s]
    public, shape = repo_fixtures(), re.compile(KEY_SHAPED.pattern.encode())
    exact, shaped, skipped = [], [], []
    for root, dirs, files in os.walk(T):
        if root == SRC:
            dirs[:] = [d for d in dirs if d not in (".venv", ".git")]
        for fn in files:
            p = os.path.join(root, fn)
            # O_NONBLOCK, then fstat: the suite leaves FIFOs in --basetemp, and a plain open()
            # of one blocks forever (the 2026-11-12 run hung here after teardown and revoke).
            try:
                fd = os.open(p, os.O_RDONLY | os.O_NONBLOCK | os.O_NOFOLLOW)
            except OSError as exc:
                skipped.append(f"{p}: {type(exc).__name__}")
                continue
            with os.fdopen(fd, "rb") as fh:
                mode = os.fstat(fd).st_mode
                if not stat.S_ISREG(mode):
                    skipped.append(f"{p}: {stat.filemode(mode)}")
                    continue
                data = fh.read()
            if any(n in data for n in needles):
                exact.append(p)
            if any(m.group() not in public for m in shape.finditer(data)):
                shaped.append(p)
    emit(step="leak_scan", exact_value_files=exact, key_shaped_files=shaped,
         repo_fixtures=len(public), skipped_non_regular=skipped)
    judge(not exact and not shaped, "a secret or key-shaped string reached a file")


def main() -> int:
    sha = subprocess.run(["git", "-C", SRC, "rev-parse", "HEAD"], capture_output=True, text=True).stdout.strip()
    emit(step="start", utc=STAMP, host="omarchy", src_sha=sha, port=PORT)
    ctx: dict = {"seen": {}}
    try:
        run(ctx)
    except Exception as exc:  # recorded; teardown and revoke still run
        judge(False, f"exception: {type(exc).__name__}: {scrub(str(exc))[:300]}")
    finally:
        stop_service(ctx)
        # verify_log always runs once the service started: its accepted ids drive teardown.
        for check, needed in ((verify_log, "svc"), (verify_state, "sid")):
            if ctx.get(needed):
                try:
                    check(ctx)
                except Exception as exc:
                    judge(False, f"{check.__name__}: {type(exc).__name__}: {scrub(str(exc))[:300]}")
        try:
            teardown_and_revoke(ctx)
        finally:
            try:
                leak_scan()
            except Exception as exc:  # a scan that cannot finish is a failed scan, not a lost summary
                judge(False, f"leak_scan: {type(exc).__name__}: {scrub(str(exc))[:300]}")
            emit(step="summary", pass_=not FAILS, fails=FAILS, finished=time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime()))
            RECEIPT.close()
    return 0 if not FAILS else 1


if __name__ == "__main__":
    sys.exit(main())
